engine.addSprite(coin); selectedSpriteState.id = player.id; } function getZelosTheme() { const themes = window.Blockly && window.Blockly.Themes ? window.Blockly.Themes : {}; return themes.Zelos || themes.Classic || {}; } function renderSpriteList() { spriteListEl.innerHTML = ''; for (const sprite of engine.sprites) { const card = document.createElement('div'); card.className = 'sprite-card'; if (sprite.id === selectedSpriteState.id) { card.classList.add('selected'); }); } function renderCostumeCanvas() { const sprite = getSelectedSprite(); if (!sprite) { costumeCtx.clearRect(0, 0, costumeCanvas.width, costumeCanvas.height); return; } costumeCtx.clearRect(0, 0, costumeCanvas.width, costumeCanvas.height); } const rect = costumeCanvas.getBoundingClientRect(); const scaleX = costumeCanvas.width / rect.width; const scaleY = costumeCanvas.height / rect.height; const x = (event.clientX - rect.left) * scaleX; const y = (event.clientY - rect.top) * scaleY; costumeCtx.lineCap = 'round'; costumeCtx.lineJoin = 'round'; costumeCtx.lineWidth = Number(brushSizeInput.value); if (activeCanvasTool === 'eraser') { costumeCtx.globalCompositeOperation = 'destination-out'; saveCurrentCostume(); } function renderSoundList() { const sprite = getSelectedSprite(); if (!sprite) { soundListEl.innerHTML = ''; return; } soundListEl.innerHTML = ''; if (!sprite.sounds || sprite.sounds.length === 0) { const empty = document.createElement('div'); const sprite = new Sprite(`Sprite ${count}`); sprite.x = (Math.random() - 0.5) * 220; sprite.y = (Math.random() - 0.5) * 160; sprite.color = ['#4ecdc4', '#f7b801', '#ff8e6e', '#81c784'][count % 4]; engine.addSprite(sprite); } function changeSelectedSpriteColor(event) { const sprite = getSelectedSprite(); if (!sprite) return; sprite.color = event.target.value; renderSpriteList(); renderCostumeCanvas(); } function changeSelectedSpriteSize(event) { const sprite = getSelectedSprite(); } }); } makeStarterSprites(); renderSpriteList(); selectSprite(engine.sprites[0].id); setActiveTab('code'); renderCostumeCanvas(); renderSoundList(); bindEvents(); engine.render(); window.engine = engine;
